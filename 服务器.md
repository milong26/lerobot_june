  --eval.n_episodes=10 \
  --eval_freq=1000 \

outputs/tinyvla_real_bushing/
└── checkpoints/
    ├── 001000/                    # 训练步骤目录（根据 total_steps 格式化）
    │   ├── pretrained_model/      # 预训练模型文件
    │   │   ├── config.json         # TinyVLA 策略配置
    │   │   ├── model.safetensors  # 模型权重（LoRA adapter 或完整模型）
    │   │   ├── train_config.json  # 完整训练配置
    │   │   ├── policy_preprocessor.json      # 预处理器配置
    │   │   ├── policy_preprocessor_step_*.safetensors  # 预处理器状态
    │   │   ├── policy_postprocessor.json     # 后处理器配置
    │   │   └── policy_postprocessor_step_*.safetensors # 后处理器状态
    │   └── training_state/       # 训练状态
    │       ├── optimizer_param_groups.json   # 优化器参数组
    │       ├── optimizer_state.safetensors   # 优化器状态
    │       ├── rng_state.safetensors         # 随机数生成器状态
    │       ├── scheduler_state.json          # 学习率调度器状态
    │       └── training_step.json            # 当前训练步骤 {"step": 1000}
    │
    └── last/                      # 最新检查点的软链接
        ├── pretrained_model/
        └── training_state/
